-- 1. الجداول create table public.profiles ( id uuid primary key references auth.users(id) on delete cascade, full_name text not null, role text not null check (role in ('teacher','student')), class_id uuid, created_at timestamptz not null default now() ); create table public.classes ( id uuid primary key default gen_random_uuid(), name text not null, code text not null unique, teacher_id uuid not null references auth.users(id) on delete cascade, created_at timestamptz not null default now() ); alter table profiles add constraint profiles_class_id_fkey foreign key (class_id) references classes(id) on delete set null; create table public.tasks ( id uuid primary key default gen_random_uuid(), title text not null, description text, deadline timestamptz not null, class_id uuid references classes(id) on delete set null, teacher_id uuid not null references auth.users(id) on delete cascade, closed boolean not null default false, deleted_at timestamptz, created_at timestamptz not null default now() ); create table public.submissions ( id uuid primary key default gen_random_uuid(), task_id uuid not null references tasks(id) on delete cascade, student_id uuid not null references auth.users(id) on delete cascade, class_id uuid references classes(id) on delete set null, note text, teacher_note text, status text not null default 'pending' check (status in ('pending','reviewed','accepted','rejected')), submitted_at timestamptz not null default now(), reviewed_at timestamptz ); create table public.submission_files ( id uuid primary key default gen_random_uuid(), submission_id uuid not null references submissions(id) on delete cascade, file_path text not null, file_name text not null, file_size bigint not null, file_type text not null, created_at timestamptz not null default now() ); -- 2. تفعيل RLS والصلاحيات (سياسات الأمان) alter table public.profiles enable row level security; alter table public.classes enable row level security; alter table public.tasks enable row level security; alter table public.submissions enable row level security; alter table public.submission_files enable row level security; create or replace function public.is_teacher() returns boolean as $$ select coalesce((select role = 'teacher' from public.profiles where id = auth.uid()), false); $$ language sql stable security definer; create or replace function public.my_class_id() returns uuid as $$ select class_id from public.profiles where id = auth.uid(); $$ language sql stable security definer; create policy "profiles_select" on profiles for select using (id = auth.uid() or (public.is_teacher() and class_id in (select id from classes where teacher_id = auth.uid()))); create policy "profiles_update" on profiles for update using (id = auth.uid()) with check (id = auth.uid() and role = (select role from profiles where id = auth.uid())); create policy "classes_select" on classes for select using (teacher_id = auth.uid() or id = public.my_class_id()); create policy "classes_insert" on classes for insert with check (teacher_id = auth.uid() and public.is_teacher()); create policy "classes_update" on classes for update using (teacher_id = auth.uid() and public.is_teacher()); create policy "classes_delete" on classes for delete using (teacher_id = auth.uid() and public.is_teacher()); create policy "tasks_select" on tasks for select using (deleted_at is null and (teacher_id = auth.uid() or class_id = public.my_class_id())); create policy "tasks_insert" on tasks for insert with check (teacher_id = auth.uid() and public.is_teacher()); create policy "tasks_update" on tasks for update using (teacher_id = auth.uid() and public.is_teacher()); create policy "tasks_delete" on tasks for delete using (teacher_id = auth.uid() and public.is_teacher()); create policy "submissions_select" on submissions for select using (student_id = auth.uid() or task_id in (select id from tasks where teacher_id = auth.uid())); create policy "submissions_insert" on submissions for insert with check (student_id = auth.uid() and class_id = public.my_class_id() and task_id in (select id from tasks where class_id = public.my_class_id() and deleted_at is null and closed = false and deadline > now())); create policy "submissions_update" on submissions for update using (task_id in (select id from tasks where teacher_id = auth.uid())) with check (task_id in (select id from tasks where teacher_id = auth.uid())); create policy "files_select" on submission_files for select using (submission_id in (select id from submissions where student_id = auth.uid() or task_id in (select id from tasks where teacher_id = auth.uid()))); create policy "files_insert" on submission_files for insert with check (submission_id in (select id from submissions where student_id = auth.uid())); -- 3. إنشاء البروفايل تلقائياً عند تسجيل مستخدم جديد create or replace function public.handle_new_user() returns trigger as $$ begin insert into public.profiles (id, full_name, role, class_id) values ( new.id, coalesce(new.raw_user_meta_data->>'full_name', 'مستخدم'), coalesce(new.raw_user_meta_data->>'role', 'student'), case when new.raw_user_meta_data->>'class_code' is not null then (select id from public.classes where code = new.raw_user_meta_data->>'class_code') else null end ); return new; end; $$ language plpgsql security definer; create trigger on_auth_user_created after insert on auth.users for each row execute function public.handle_new_user();
